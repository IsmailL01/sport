       ```typescript
       // apps/mobile-rn/src/foreground/notification.ts
       // Фоновая нотификация для активной сессии записи трека.
       // Динамическое обновление контента каждые ~5 секунд через SessionManager.recordingTick.
       // VERIFY-FOREGROUND-NOTIFICATION-API: confirm expo-notifications SDK 54 schema is current
       import * as Notifications from 'expo-notifications';
       import { Platform } from 'react-native';
       import { useSessionStore } from '../state/activity'; // existing Zustand store

       const NOTIFICATION_ID = 'recording-status'; // de-dupes; replaces previous
       const CHANNEL_ID = 'recording'; // Android notification channel

       export async function setupForegroundChannel(): Promise<void> {
         if (Platform.OS !== 'android') return;
         await Notifications.setNotificationChannelAsync(CHANNEL_ID, {
           name: 'Запись пробежки',
           importance: Notifications.AndroidImportance.LOW, // non-intrusive ongoing
           lockscreenVisibility: Notifications.AndroidNotificationVisibility.PUBLIC,
           sound: null,
           vibrationPattern: null,
           lightColor: '#0F1419',
         });
       }

       export async function presentRecordingNotification(duration: string, distance: string): Promise<void> {
         if (Platform.OS !== 'android') return;
         await Notifications.scheduleNotificationAsync({
           identifier: NOTIFICATION_ID,
           content: {
             title: 'Running Ecosystem',
             body: `Запись пробежки активна — ${duration} • ${distance}`,
             categoryIdentifier: 'recording',
             sticky: true,
             autoDismiss: false,
             color: '#0F1419',
           },
           trigger: null, // present immediately; replaces NOTIFICATION_ID if exists
         });
       }

       export async function dismissRecordingNotification(): Promise<void> {
         if (Platform.OS !== 'android') return;
         await Notifications.dismissNotificationAsync(NOTIFICATION_ID);
       }

       /**
        * Subscribe to SessionManager.recordingTick. Called from App.tsx onMount;
        * unsubscribe on app unmount (not strictly needed — app lifecycle ends).
        */
       export function subscribeToRecordingTick(): () => void {
         setupForegroundChannel().catch(() => {/* non-fatal — channel may already exist */});
         return useSessionStore.subscribe(
           (state) => ({ duration: state.duration, distance: state.distance, status: state.status }),
           ({ duration, distance, status }) => {
             if (status === 'recording') {
               presentRecordingNotification(duration, distance).catch(() => {/* non-fatal */});
             } else {
               dismissRecordingNotification().catch(() => {/* non-fatal */});
             }
           },
           { fireImmediately: true }
         );
       }
       ```

       Then wire it from `apps/mobile-rn/App.tsx` (or wherever the root mounts):
       ```typescript
       // App.tsx — inside the root component
       useEffect(() => {
         const unsubscribe = subscribeToRecordingTick();
         return () => unsubscribe();
       }, []);
       ```

       NOTE: the exact field names from `useSessionStore` may differ (`session.points.length` vs `distance`).

       ```typescript
       // apps/mobile-rn/src/vendor/openOEMSettings.ts
       // Открывает OEM-специфичные настройки auto-start + battery-saver.
       // Try-catch fallback: при отсутствии OEM-intent на устройстве откроется generic app-details settings.
       import { startActivityAsync, ActivityAction } from 'expo-intent-launcher';
       import { Platform } from 'react-native';
       import * as Application from 'expo-application';
       import { detectVendor } from './oem';

       const APP_PACKAGE = Application.applicationId ?? 'com.runningecosystem.mobile';

       export type OEMSettingsResult = { launched: boolean; vendor: string };

       export async function openOEMAutoStartSettings(): Promise<OEMSettingsResult> {
         if (Platform.OS !== 'android') return { launched: false, vendor: 'ios' };
         const vendor = detectVendor();
         try {
           switch (vendor) {
             case 'xiaomi':
               await startActivityAsync('miui.intent.action.APP_PERM_EDITOR', {
                 extra: { extra_pkgname: APP_PACKAGE },
               });
               return { launched: true, vendor: 'xiaomi' };
             case 'samsung':
               await startActivityAsync('com.samsung.android.sm.ACTION_BATTERY');
               return { launched: true, vendor: 'samsung' };
             default:
               await startActivityAsync(ActivityAction.APPLICATION_DETAILS_SETTINGS, {
                 data: `package:${APP_PACKAGE}`,
               });
               return { launched: true, vendor: 'generic' };
           }
         } catch (e) {
           // OEM intent unavailable on this device — fall back to generic
           try {
             await startActivityAsync(ActivityAction.APPLICATION_DETAILS_SETTINGS, {
               data: `package:${APP_PACKAGE}`,
             });
             return { launched: true, vendor: 'generic-fallback' };
           } catch {
             return { launched: false, vendor: 'launch-failed' };
           }
         }
       }
       ```
