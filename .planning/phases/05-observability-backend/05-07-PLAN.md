- Drops sentry-cli alert rule codification — Sentry-side alerts via sentry.io UI per ADR-0010.
- Adds explicit niko-prod hands-off invariants (Docker network namespacing, no nginx touch, separate /opt/ directory).
- Adds Caddy binary install (not container) per RESEARCH §P17 Docker socket UID/GID concern.
- Adds `tls internal` self-signed TLS handling per D-37.

- `https://82.25.71.215:8443/grafana/` — dev UI access (basicauth), 4 dashboards loaded (Plan 05-04 ships JSONs; this plan mounts them)

<!-- 2. Caddyfile on srv1561293 — :8443 ONLY, self-signed TLS, 3 reverse-proxy paths -->

{
    # Global options
    admin off    # disable admin API
}

# Single site on :8443 — does NOT touch :80/:443 owned by existing system nginx
:8443 {
    tls internal    # self-signed via Caddy's local CA (D-37)

    # Path 1: Loki push (Alloy from prod VPS) — IP allowlist per D-36
    @allowed_loki { remote_ip 148.253.214.156/32 }
    handle_path /loki/* {
        not @allowed_loki respond 403
        reverse_proxy 127.0.0.1:3100
    }

    # Path 2: Grafana UI (dev access via basicauth)
    handle_path /grafana/* {
        basicauth {
            admin {$GRAFANA_ADMIN_PASSWORD_BCRYPT}
        }
        reverse_proxy 127.0.0.1:3000
    }

    # Path 3: Prometheus UI / API (ad-hoc queries; rarely needed)
    handle_path /prometheus/* {
        basicauth {
            admin {$GRAFANA_ADMIN_PASSWORD_BCRYPT}
        }
        reverse_proxy 127.0.0.1:9090
    }

    # Catch-all: 404
    respond 404
}
